\begin{listofsymbols}[4cm]
\item [$T$] Nombre de frames d'une séquence, égal à 30
\item [$D$] Nombre de caractéristiques par frame, égal à 225
\item [$C$] Nombre de classes du vocabulaire, égal à 10
\item [$N$] Nombre de séquences d'un ensemble de données
\item [$y_i$] Classe réelle de la séquence $i$
\item [$\hat{y}_i$] Classe prédite pour la séquence $i$
\item [$p_c$] Probabilité prédite pour la classe $c$
\item [$P_c$] Précision de la classe $c$
\item [$R_c$] Rappel de la classe $c$
\item [$F_{1,c}$] F1-score de la classe $c$
\item [$F_{1,\mathrm{macro}}$] Moyenne des F1-scores des $C$ classes
\item [$TP_c$] Nombre de vrais positifs de la classe $c$
\item [$FP_c$] Nombre de faux positifs de la classe $c$
\item [$FN_c$] Nombre de faux négatifs de la classe $c$
\item [$\mathbf{1}(\cdot)$] Fonction indicatrice, égale à 1 si la condition est vraie et à 0 sinon
\item [$\sigma$] Écart-type du bruit gaussien ajouté aux coordonnées
\item [\texttt{[30,225]}] Forme d'une séquence d'entrée du modèle
\end{listofsymbols}
